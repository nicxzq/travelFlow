### 1.4 `lib/supabase/server.ts`（新建）

Next 14 的 `cookies()` **是同步的**，不要 `await`。

```ts
import { createServerClient } from '@supabase/ssr';
import { cookies } from 'next/headers';
import type { Database } from '@/types/supabase';

export function createSupabaseServerClient() {
  const cookieStore = cookies();

  return createServerClient<Database>(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => cookieStore.getAll(),
        setAll: (list) => {
          try {
            list.forEach(({ name, value, options }) => cookieStore.set(name, value, options));
          } catch {
            // Server Component 渲染期不可写 cookie，由 middleware 承担刷新
          }
        },
      },
    },
  );
}
```

**禁止把返回值缓存到模块级变量。**

### 1.5 `lib/supabase/middleware.ts`（新建）

三个易错点已内联标注：模板页必须公开、API 不得重定向、每个提前返回的响应都要复制 cookie。

```ts
import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

const PUBLIC_PREFIXES = ['/login', '/register', '/auth', '/api/auth/hooks'];

// 精确匹配或带分隔符的前缀匹配，避免 /loginfoo、/authorize 被误判为公开
const isPublicPath = (pathname: string) =>
  pathname === '/' ||
  pathname.startsWith('/trip/') ||    // 模板详情页对 anon 开放，真实可见性由 RLS 裁决
  PUBLIC_PREFIXES.some((p) => pathname === p || pathname.startsWith(`${p}/`));

export async function updateSession(request: NextRequest) {
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => request.cookies.getAll(),
        setAll: (list) => {
          list.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          list.forEach(({ name, value, options }) => response.cookies.set(name, value, options));
        },
      },
    },
  );

  // 此行与 createServerClient 之间不得插入任何代码
  const { data: { user } } = await supabase.auth.getUser();

  const { pathname, search } = request.nextUrl;

  if (!user && !isPublicPath(pathname)) {
    // API 路由必须返回 JSON。若走 redirect，fetch 会跟随 307 拿到 HTML，
    // 且路由内的 401 分支永远执行不到。
    if (pathname.startsWith('/api/')) {
      const denied = NextResponse.json({ error: 'UNAUTHENTICATED' }, { status: 401 });
      response.cookies.getAll().forEach((c) => denied.cookies.set(c));
      return denied;
    }

    const url = request.nextUrl.clone();
    url.pathname = '/login';
    url.search = '';
    url.searchParams.set('next', `${pathname}${search}`);
    const redirect = NextResponse.redirect(url);
    response.cookies.getAll().forEach((c) => redirect.cookies.set(c));  // 不可省略
    return redirect;
  }

  return response;
}
```

**每一个提前返回的响应都必须复制 `response.cookies`**，否则刷新后的 token 在该请求上丢失——重定向与 401 是同一类故障的两半。

### 3.1 `0110_profiles.sql`

```sql
alter table public.profiles
  add column if not exists phone      text,
  add column if not exists updated_at timestamptz not null default now();

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles (id, email, phone, display_name)
  values (new.id, new.email, new.phone,
          coalesce(new.raw_user_meta_data ->> 'display_name', ''))
  on conflict (id) do nothing;
  return new;
exception when others then
  raise warning 'handle_new_user failed for %: %', new.id, sqlerrm;  -- 不静默
  return new;   -- 绝不阻断注册
end;
$$;

revoke execute on function public.handle_new_user() from public, anon, authenticated;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- 兜底通道：profiles 无 INSERT 策略，应用侧必须经此函数补齐，
-- 否则触发器失败的用户会永久卡在「有 auth.users 无 profiles」，建行程时 23503。
-- 无参数：只可能写调用者自己那行。
create or replace function public.ensure_profile()
returns void
language plpgsql
security definer
set search_path = ''
as $$
begin
  insert into public.profiles (id, email, phone, display_name)
  select u.id, u.email, u.phone,
         coalesce(u.raw_user_meta_data ->> 'display_name', '')
    from auth.users u
   where u.id = (select auth.uid())
  on conflict (id) do nothing;
end;
$$;

revoke execute on function public.ensure_profile() from public, anon;
grant  execute on function public.ensure_profile() to authenticated;
```

### 3.7 `0200_grants.sql`（必须先于 0300）

**必须先 REVOKE 再 GRANT。** 托管 Supabase 在初始化时执行过 `alter default privileges in schema public grant all on tables to anon, authenticated`，`public` 下的表天然带写权限——只补 GRANT 会让 anon 保留 INSERT / UPDATE / DELETE，使「策略写错」的后果从拒绝退化为放行。

```sql
revoke all on public.trips, public.days, public.events, public.memories,
              public.profiles, public.trip_documents, public.todos,
              public.destination_map_points, public.ai_usage_quotas,
              public.otp_rate_limits
  from anon, authenticated;

-- anon：仅模板可见的只读面
grant select on public.trips, public.days, public.events, public.trip_documents,
                public.todos, public.destination_map_points
  to anon;

grant select, insert, update, delete
  on public.trips, public.days, public.events, public.memories,
     public.trip_documents, public.todos, public.destination_map_points
  to authenticated;
grant select, update on public.profiles to authenticated;

-- 配额只读：写入唯一通道是 consume_ai_quota()。
-- 计数器绝不能对被计数的角色开放写权限，否则用户可自行 PATCH count = 0。
grant select on public.ai_usage_quotas to authenticated;
```

### 3.8 `0250_helpers.sql`

⚠️ 编号必须小于 `0300`：`0300` 里的策略引用这两个函数，按文件名顺序执行时函数必须先存在。

```sql
create or replace function public.is_trip_owner(p_trip_id uuid)
returns boolean language sql stable security invoker set search_path = ''
as $$
  select exists (
    select 1 from public.trips t
    where t.id = p_trip_id
      and t.user_id = (select auth.uid())
      and t.is_template = false
  );
$$;

create or replace function public.is_trip_readable(p_trip_id uuid)
returns boolean language sql stable security invoker set search_path = ''
as $$
  select exists (
    select 1 from public.trips t
    where t.id = p_trip_id
      and (t.is_template = true or t.user_id = (select auth.uid()))
  );
$$;

grant execute on function public.is_trip_owner(uuid)    to authenticated;
grant execute on function public.is_trip_readable(uuid) to anon, authenticated;
```

⚠️ **这两个函数必须保留 `PUBLIC` 的默认 EXECUTE 授权，不要 revoke**——与 §8.3 的 RPC 规则相反。它们被 `to anon` 的 SELECT 策略调用，一旦 revoke，anon 访问模板页会直接报错。

⚠️ `set search_path = ''` 写入 `pg_proc.proconfig`，而 `inline_function()` 对 `proconfig` 非空的函数**直接否决内联**。因此这两个函数按行调用而非展开，此处以安全性优先接受该成本。不要在注释里写「为了内联」——那是错的。

### 3.9 `0300_rls_and_policies.sql`

`enable RLS` 与 `create policy` **必须在同一事务**，否则中间窗口里所有表 RLS 已开、零策略，全站不可用。每条策略前置 `drop policy if exists` 以便反复调试。

```sql
begin;

alter table public.profiles        enable row level security;
alter table public.trips           enable row level security;
alter table public.days            enable row level security;
alter table public.events          enable row level security;
alter table public.memories        enable row level security;
alter table public.trip_documents  enable row level security;
alter table public.ai_usage_quotas enable row level security;
-- todos / destination_map_points 已在 itinerary-management-extension.sql 启用

-- ---------- profiles：无 INSERT/DELETE 策略，只允许 definer 函数写入 ----------
drop policy if exists "profiles select self" on public.profiles;
create policy "profiles select self" on public.profiles
  for select to authenticated using (id = (select auth.uid()));

drop policy if exists "profiles update self" on public.profiles;
create policy "profiles update self" on public.profiles
  for update to authenticated
  using (id = (select auth.uid())) with check (id = (select auth.uid()));

-- ---------- trips：逐命令写，不用 FOR ALL ----------
drop policy if exists "trips select" on public.trips;
create policy "trips select" on public.trips
  for select to anon, authenticated
  using (is_template = true or user_id = (select auth.uid()));

-- slug 钉为 null：防止用户抢占尚未 seed 的模板 slug，导致 seed 脚本 23505
drop policy if exists "trips insert" on public.trips;
create policy "trips insert" on public.trips
  for insert to authenticated
  with check (user_id = (select auth.uid()) and is_template = false and slug is null);

drop policy if exists "trips update" on public.trips;
create policy "trips update" on public.trips
  for update to authenticated
  using      (user_id = (select auth.uid()) and is_template = false)
  with check (user_id = (select auth.uid()) and is_template = false and slug is null);

drop policy if exists "trips delete" on public.trips;
create policy "trips delete" on public.trips
  for delete to authenticated
  using (user_id = (select auth.uid()) and is_template = false);

-- ---------- days ----------
drop policy if exists "days select" on public.days;
create policy "days select" on public.days
  for select to anon, authenticated using (public.is_trip_readable(trip_id));
drop policy if exists "days insert" on public.days;
create policy "days insert" on public.days
  for insert to authenticated with check (public.is_trip_owner(trip_id));
drop policy if exists "days update" on public.days;
create policy "days update" on public.days
  for update to authenticated
  using (public.is_trip_owner(trip_id)) with check (public.is_trip_owner(trip_id));
drop policy if exists "days delete" on public.days;
create policy "days delete" on public.days
  for delete to authenticated using (public.is_trip_owner(trip_id));

-- ---------- events：写入必须校验 day 归属，否则可把 event 挂到他人的 day 上 ----------
drop policy if exists "events select" on public.events;
create policy "events select" on public.events
  for select to anon, authenticated using (public.is_trip_readable(trip_id));

drop policy if exists "events insert" on public.events;
create policy "events insert" on public.events
  for insert to authenticated
  with check (
    public.is_trip_owner(trip_id)
    and exists (select 1 from public.days d where d.id = day_id and d.trip_id = events.trip_id)
  );

drop policy if exists "events update" on public.events;
create policy "events update" on public.events
  for update to authenticated
  using (public.is_trip_owner(trip_id))
  with check (
    public.is_trip_owner(trip_id)
    and exists (select 1 from public.days d where d.id = day_id and d.trip_id = events.trip_id)
  );

drop policy if exists "events delete" on public.events;
create policy "events delete" on public.events
  for delete to authenticated using (public.is_trip_owner(trip_id));

-- ---------- trip_documents ----------
drop policy if exists "trip docs select" on public.trip_documents;
create policy "trip docs select" on public.trip_documents
  for select to anon, authenticated using (public.is_trip_readable(trip_id));
drop policy if exists "trip docs write" on public.trip_documents;
create policy "trip docs write" on public.trip_documents
  for all to authenticated
  using (public.is_trip_owner(trip_id)) with check (public.is_trip_owner(trip_id));

-- ---------- memories：event_id 归属校验与 events 同理，不可只做一半 ----------
drop policy if exists "memories select" on public.memories;
create policy "memories select" on public.memories
  for select to authenticated using (public.is_trip_owner(trip_id));

drop policy if exists "memories write" on public.memories;
create policy "memories write" on public.memories
  for all to authenticated
  using (public.is_trip_owner(trip_id))
  with check (
    public.is_trip_owner(trip_id)
    and (event_id is null or exists (
      select 1 from public.events e where e.id = event_id and e.trip_id = memories.trip_id))
  );

-- ---------- 既有扩展表：补模板可读，否则 anon 打开模板页拿 42501 ----------
drop policy if exists "todos read template or owner" on public.todos;
create policy "todos read template or owner" on public.todos
  for select to anon, authenticated using (public.is_trip_readable(trip_id));

drop policy if exists "map points read template or owner" on public.destination_map_points;
create policy "map points read template or owner" on public.destination_map_points
  for select to anon, authenticated using (public.is_trip_readable(trip_id));

-- 既有的 todos / map points 写策略保持不变：其内联 EXISTS 要求 trips.user_id = auth.uid()，
-- 模板 user_id IS NULL 使其为 false，模板天然不可写。

-- ---------- ai_usage_quotas：只读 ----------
drop policy if exists "quota self" on public.ai_usage_quotas;
drop policy if exists "quota self read" on public.ai_usage_quotas;
create policy "quota self read" on public.ai_usage_quotas
  for select to authenticated using (user_id = (select auth.uid()));

commit;
```

### 4.1 `0400_rpc_create.sql`

```sql
create or replace function public.create_trip_from_itinerary(
  p_payload       jsonb,
  p_import_key    text default null,
  p_source_prompt text default null
) returns jsonb
language plpgsql
security invoker
set search_path = ''
as $$
declare
  v_uid        uuid := (select auth.uid());
  v_trip_id    uuid;
  v_day_id     uuid;
  v_day        jsonb;
  v_event      jsonb;
  v_sort       int;
  v_day_no     int := 0;
  v_constraint text;
begin
  if v_uid is null then
    raise exception 'unauthenticated' using errcode = '42501';
  end if;

  if p_import_key is not null then
    select id into v_trip_id from public.trips
      where user_id = v_uid and client_import_key = p_import_key;
    if found then
      return jsonb_build_object('trip_id', v_trip_id, 'created', false);
    end if;
  end if;

  if pg_column_size(p_payload) > 512 * 1024 then     -- jsonb 二进制大小，非文本长度
    raise exception 'payload too large' using errcode = '22023';
  end if;

  -- is distinct from：键缺失时 -> 返回 SQL NULL -> `<> 'array'` 求值为 NULL
  -- -> plpgsql 的 IF NULL 按 false 处理 -> 异常不抛 -> 静默建出空行程。
  if jsonb_typeof(p_payload -> 'detailedItinerary') is distinct from 'array'
     or jsonb_array_length(p_payload -> 'detailedItinerary') not between 1 and 30 then
    raise exception 'invalid itinerary shape' using errcode = '22023';
  end if;

  insert into public.trips (user_id, title, destination, status, is_template, client_import_key)
  values (v_uid,
          left(coalesce(p_payload ->> 'tripTitle', '未命名行程'), 200),
          left(coalesce(p_payload #>> '{tripHighlights,destination}', ''), 200),
          'planning', false, p_import_key)
  returning id into v_trip_id;

  for v_day in select * from jsonb_array_elements(p_payload -> 'detailedItinerary') loop
    if jsonb_typeof(v_day -> 'events') is distinct from 'array'
       or jsonb_array_length(v_day -> 'events') > 50 then
      raise exception 'invalid events shape' using errcode = '22023';
    end if;

    v_day_no := v_day_no + 1;

    insert into public.days (trip_id, day_index, date, summary, theme)
    values (
      v_trip_id,
      -- fallback 用循环序号而非常量 1，否则缺 dayIndex 时全部塌成第 1 天
      case when coalesce(v_day ->> 'dayIndex', '') ~ '^\d+$'
           then (v_day ->> 'dayIndex')::int else v_day_no end,
      case when coalesce(v_day ->> 'date', '') ~ '^\d{4}-\d{2}-\d{2}$'
           then (v_day ->> 'date')::date else null end,
      left(coalesce(v_day ->> 'theme', ''), 500),
      left(coalesce(v_day ->> 'theme', ''), 200))
    returning id into v_day_id;

    v_sort := 0;
    for v_event in select * from jsonb_array_elements(v_day -> 'events') loop
      insert into public.events (trip_id, day_id, title, description, category,
                                 location_name, start_time, lat, lng,
                                 transport_mode, sort_order, status)
      values (
        v_trip_id, v_day_id,
        left(coalesce(v_event ->> 'title', '未命名'), 200),
        left(coalesce(v_event ->> 'description', ''), 4000),
        case when v_event ->> 'category' in ('spot','food','hotel','transport','custom')
             then v_event ->> 'category' else 'custom' end,
        left(coalesce(v_event ->> 'location', ''), 200),
        -- 正则收紧到 00:00-23:59。原 '\d{1,2}:\d{2}' 会匹配 '25:99' 然后 ::time 抛 22008，
        -- 一个脏字段就让整单回滚。
        nullif(substring(coalesce(v_event ->> 'time', '')
                         from '(?:[01][0-9]|2[0-3]|[0-9]):[0-5][0-9]'), '')::time,
        case when coalesce(v_event #>> '{geo,lat}', '') ~ '^-?\d+(\.\d+)?$'
              and (v_event #>> '{geo,lat}')::double precision between -90 and 90
             then (v_event #>> '{geo,lat}')::double precision else null end,
        case when coalesce(v_event #>> '{geo,lng}', '') ~ '^-?\d+(\.\d+)?$'
              and (v_event #>> '{geo,lng}')::double precision between -180 and 180
             then (v_event #>> '{geo,lng}')::double precision else null end,
        case when v_event ->> 'transport' in ('drive','flight','walk')
             then v_event ->> 'transport' else null end,
        v_sort, 'planned'
      );
      v_sort := v_sort + 1;
    end loop;
  end loop;

  insert into public.trip_documents (trip_id, schema_version, source_prompt, payload)
  values (v_trip_id, 1, left(coalesce(p_source_prompt, ''), 4000), p_payload);

  return jsonb_build_object('trip_id', v_trip_id, 'created', true);

exception
  when unique_violation then
    -- 只认幂等键这一个约束。无条件吞掉会把 days_trip_day_index_key 之类的冲突
    -- 伪装成「已导入过」，并返回 {"trip_id": null, "created": false}
    -- —— 调用方会把 null 当成功落盘，该记录从此永不重试。
    if p_import_key is null then
      raise;
    end if;

    get stacked diagnostics v_constraint = constraint_name;
    if v_constraint is distinct from 'trips_user_import_key' then
      raise;
    end if;

    select id into v_trip_id from public.trips
      where user_id = v_uid and client_import_key = p_import_key;
    if v_trip_id is null then
      raise;
    end if;

    return jsonb_build_object('trip_id', v_trip_id, 'created', false);
end;
$$;

revoke execute on function public.create_trip_from_itinerary(jsonb, text, text) from public, anon;
grant  execute on function public.create_trip_from_itinerary(jsonb, text, text) to authenticated;
```

⚠️ `'2026-02-30'` 这类合法形状但非法日期仍会抛错。若要绝对安全需包一层 `exception when others then return null` 的辅助函数，代价是引入子事务；按 30 天上限的量级可接受，视实测决定。

### 4.2 `0410_rpc_copy_template.sql`

**必须用 plpgsql 循环，不能用 data-modifying CTE。** 原因见设计文档 §8.4：CTE 与主查询共享 snapshot 与 CommandId，`events` INSERT 策略里的 `exists (select 1 from public.days ...)` 是 SubPlan、要真去扫堆表，看不到同语句 CTE 刚插入的 day，每次调用必然在第一条 event 上抛 `42501`。

```sql
create or replace function public.copy_template_trip(p_source uuid)
returns uuid
language plpgsql
security invoker
set search_path = ''
as $$
declare
  v_uid     uuid := (select auth.uid());
  v_trip_id uuid;
  v_day     record;
  v_new_day uuid;
begin
  if v_uid is null then
    raise exception 'unauthenticated' using errcode = '42501';
  end if;

  -- 不可省略：漏掉即成越权读取通道，任何登录用户都能复制他人私有行程
  if not exists (select 1 from public.trips where id = p_source and is_template) then
    raise exception 'not a template' using errcode = '42501';
  end if;

  insert into public.trips (user_id, title, destination, start_date, end_date,
                            status, is_template, template_source_id, cover_url)
  select v_uid, t.title, t.destination, t.start_date, t.end_date,
         'planning', false, t.id, t.cover_url
    from public.trips t where t.id = p_source
  returning id into v_trip_id;

  -- 每个 day 独立成句：SPI 在语句之间调用 CommandCounterIncrement()，
  -- events 的 RLS 子查询因此能看到刚插入的 day。
  for v_day in
    select d.id, d.day_index, d.date, d.summary, d.theme
      from public.days d where d.trip_id = p_source order by d.day_index
  loop
    insert into public.days (trip_id, day_index, date, summary, theme)
    values (v_trip_id, v_day.day_index, v_day.date, v_day.summary, v_day.theme)
    returning id into v_new_day;

    -- 完整列清单：tags / image_url / story 是模板自带的 overlay 数据，
    -- 漏拷会让首次复制的用户永久少掉标签与图片。
    insert into public.events (trip_id, day_id, title, description, category, location_name,
                               address, start_time, end_time, is_time_flexible,
                               lat, lng, transport_mode, legacy_key,
                               image_url, story, tags,
                               sort_order, status, navigation_url, external_link)
    select v_trip_id, v_new_day, e.title, e.description, e.category, e.location_name,
           e.address, e.start_time, e.end_time, e.is_time_flexible,
           e.lat, e.lng, e.transport_mode, e.legacy_key,
           e.image_url, e.story, e.tags,
           e.sort_order, 'planned', e.navigation_url, e.external_link
      from public.events e
     where e.day_id = v_day.id
     order by e.sort_order, e.start_time nulls last;
  end loop;

  insert into public.trip_documents (trip_id, schema_version, source_prompt, payload)
  select v_trip_id, td.schema_version, td.source_prompt, td.payload
    from public.trip_documents td where td.trip_id = p_source;

  return v_trip_id;
end;
$$;

revoke execute on function public.copy_template_trip(uuid) from public, anon;
grant  execute on function public.copy_template_trip(uuid) to authenticated;
```

### 4.3 `0420_rpc_replace.sql`

```sql
create or replace function public.replace_trip_content(p_trip_id uuid, p_payload jsonb)
returns jsonb
language plpgsql
security invoker
set search_path = ''
as $$
declare
  v_day     jsonb;
  v_event   jsonb;
  v_day_id  uuid;
  v_sort    int;
  v_day_no  int := 0;
begin
  if not public.is_trip_owner(p_trip_id) then
    raise exception 'forbidden' using errcode = '42501';
  end if;

  if jsonb_typeof(p_payload -> 'detailedItinerary') is distinct from 'array'
     or jsonb_array_length(p_payload -> 'detailedItinerary') not between 1 and 30 then
    raise exception 'invalid itinerary shape' using errcode = '22023';
  end if;

  -- 依赖 0135 的 ON DELETE CASCADE：days -> events -> memories
  delete from public.days where trip_id = p_trip_id;

  -- 重建逻辑与 create_trip_from_itinerary 完全一致，此处从略；
  -- 实现时抽成共享的内部函数 public._insert_itinerary_days(p_trip_id, p_payload)
  -- 供两者调用，避免两份漂移的解析逻辑。

  update public.trips
     set title       = left(coalesce(p_payload ->> 'tripTitle', title), 200),
         destination = left(coalesce(p_payload #>> '{tripHighlights,destination}', destination), 200)
   where id = p_trip_id;

  insert into public.trip_documents (trip_id, schema_version, payload)
  values (p_trip_id, 1, p_payload)
  on conflict (trip_id) do update
    set payload        = excluded.payload,
        schema_version = public.trip_documents.schema_version + 1;

  return jsonb_build_object('trip_id', p_trip_id, 'replaced', true);
end;
$$;

revoke execute on function public.replace_trip_content(uuid, jsonb) from public, anon;
grant  execute on function public.replace_trip_content(uuid, jsonb) to authenticated;
```

⚠️ `delete from days` 会连带删除该行程所有 event 上的 `image_url` / `story` / `tags`。若「重新生成」需保留 overlay，实现时必须先按 `legacy_key` 或 `(day_index, sort_order)` 快照，重建后回填——这一点须在 Stage 5 的 UI 上向用户明示。

```sql
create or replace function public.consume_ai_quota(p_limit int default 20)
returns int
language plpgsql
security definer
set search_path = ''

  v_uid   uuid := (select auth.uid());
```

### 4.5 `lib/validation/itinerary-schema.ts`

**`.passthrough()` 只作用于它所在的那一层 `z.object`。** 嵌套对象默认是 `strip` 模式——`lib/domain/trip.ts:98-99` 的 `transport` 与 `tips` 会被静默剥除，而这两个字段恰恰是规范化表里没有、本该由 `trip_documents.payload` 承载的内容，剥掉后两边都不存在，数据直接丢失。每一层都要 `passthrough`。

```ts
import { z } from 'zod';

const EventSchema = z.object({
  time: z.string().max(20).optional().default(''),
  title: z.string().min(1).max(200),
  category: z.enum(['spot', 'food', 'hotel', 'transport', 'custom']).catch('custom'),
  description: z.string().max(4000).optional().default(''),
  location: z.string().max(200).optional().default(''),
  transport: z.string().max(500).optional().default(''),
  tips: z.string().max(2000).optional().default(''),
  geo: z.object({ lat: z.number().min(-90).max(90), lng: z.number().min(-180).max(180) }).optional(),
}).passthrough();

const DaySchema = z.object({
  dayIndex: z.number().int().min(1).max(60),
  date: z.string().max(20).optional(),
  theme: z.string().max(500).optional(),
  events: z.array(EventSchema).max(50),
}).passthrough();

export const ItinerarySchema = z.object({
  tripTitle: z.string().min(1).max(200),
  tripHighlights: z.object({ destination: z.string().max(200) }).partial().passthrough().optional(),
  detailedItinerary: z.array(DaySchema).min(1).max(30),
}).passthrough();
```

⚠️ AI 输出的 `transport` 是自由文本，只有恰好落在 `drive` / `flight` / `walk` 时才写入 `events.transport_mode`，否则为 NULL；原文始终保留在 payload 里。

### 5.5 overlay 的 localStorage key 迁移

改造前 `getTripDetails('shanxi-loop-2026')` 返回的对象 `id === 'shanxi-loop-2026'`，因此现存用户的 overlay 存在 **slug key** 下（`lib/journey/journey-store.ts:8`）。seed 入库后 `trip.id` 变成 uuid，工作台会去读一个空的 uuid key——**用户已有的图片、游记、标签会在页面上直接消失**（数据仍在 localStorage，但无人读取）。

### 6.2 算法

```
onLoginSuccess():
  await supabase.rpc('ensure_profile')              // 先补 profile，否则建行程 23503

  items   = safeParse(localStorage['travelflow_local_itinerary_list_v1']) ?? []
  state   = safeParse(localStorage['travelflow:import-state:v1']) ?? {}
  pending = items.filter(i => !state[keyOf(i)])
  if pending.length === 0: return                    // 静默跳过

  展示 pending 列表，等待用户勾选并点击「导入到我的账号」   // 绝不自动导入

  for each selected (串行):
    try:
      parsed = ItinerarySchema.parse(item.itinerary)
      res    = await importLocalItinerary({ payload: parsed,
                                            importKey: keyOf(item),
                                            sourcePrompt: item.sourcePrompt })
      if (!res?.trip_id) throw new Error('IMPORT_NO_TRIP_ID')   // 绝不把 null 当成功
      state[keyOf(item)] = { tripId: res.trip_id, at: Date.now() }
      localStorage['travelflow:import-state:v1'] = JSON.stringify(state)   // 每条立即落盘
    catch e:
      failures.push({ title: item.itinerary.tripTitle, reason: mapError(e) })

  渲染「已导入 N / M」+ 失败原因 + 「重试失败项」按钮
  绝不删除原始 localStorage 数据；清理由用户点独立按钮触发
```

```ts
keyOf = (i) => `local-v1:${sha256(`${i.id}|${i.createdAt}|${i.itinerary.tripTitle}`).slice(0, 32)}`
```

`res.trip_id` 的非空断言不可省略：它是 Stage 4.1 异常处理器修复的客户端配套。一旦把 `null` 当成功写进 `import-state`，该条记录会被永久跳过且永不重试，直接摧毁「全部重试永远安全」的承诺。

**串行而非并发**：并发放大 unique_violation 概率，跨境链路上也快不了多少。因 RPC 幂等，「全部重试」永远安全。

### 6.3 Overlay 迁移（仅模板复制场景）

`copy_template_trip` 返回新 trip_id 后，客户端查 `select id, legacy_key from events where trip_id = <新id>` 得到映射，再读 `travelflow:journey-overlay:{原slug}:v1`，按 `legacy_key` 换成新 event uuid，调 `applyJourneyOverlay` 写入 `events.image_url / story / tags`。

## 附录 B · 审计确认无误的部分

以下经双路独立审计确认正确，实施时不要"顺手优化"：

- `security invoker` 的 `create_trip_from_itinerary` 内四次 INSERT 全部能通过 RLS——plpgsql 语句之间有 `CommandCounterIncrement()`，后一条能看见前一条的效果。这与 §4.2 的 CTE 失败恰成对照，**唯一区别就是是否写在同一条语句里**。
- `handle_new_user` 的 `exception when others then return new` 确实能防止 `auth.users` 的 INSERT 回滚（plpgsql exception 块开子事务，回滚到 savepoint 后继续）。
- 对触发器函数 `revoke execute from public, anon, authenticated` 是正确的：EXECUTE 只在 `CREATE TRIGGER` 时检查，触发时不做 ACL 检查，owner 保留自身权限。
- 模板不可写是双重锁死：三条写策略都带 `is_template = false`，且 CHECK 约束令 `is_template = true ⇒ user_id IS NULL`，与 `user_id = auth.uid()` 互斥。
- anon 的 SELECT 策略不泄漏私有数据：`auth.uid()` 为 NULL 时 `user_id = NULL` 求值为 NULL 而非 true。
- dollar-quoted 函数体内 `'\d{1,2}:\d{2}'` 的转义正确（dollar-quoting 不处理反斜杠，`standard_conforming_strings=on` 下原样传给正则引擎）。收紧正则是为了值域，不是为了转义。
- 文档中不存在 `left()` 吃 NULL 的路径，每处都有 `coalesce` 包裹。
- Next 14.2.30：`cookies()` 同步、`middleware.ts` 于仓库根（无 `src/`）、matcher 正则有效、`createBrowserClient` / `createServerClient` 签名匹配 `@supabase/ssr ≥ 0.5`。
- 迁移顺序正确：建表(0140) → 授权(0200) → 辅助函数(0250) → 启用 RLS 与策略(0300，同一事务) → RPC(0400+)。
